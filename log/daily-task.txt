@app.post("/api/generate")
async def generate_task(request: GenerateRequest, background_tasks: BackgroundTasks):
    """
    Generate app from user requirements.
    创建任务后立即在后台运行 generator_service，返回 task_id 供前端轮询进度。
    """
    try:
        user_requirements = (request.user_requirements or "").strip()
        if not user_requirements:
            return {"status": "error", "message": "user_requirements is required"}

        task_id = task_manager.create_task()
        background_tasks.add_task(run_app_generation_task, task_id, user_requirements)
        return {
            "status": "success",
            "message": "Generation task created",
            "task_id": task_id
        }
    except Exception as e:
        return {"status": "error", "message": str(e)}

          const filteredTasks = this.getFilteredTasks();
          this.task_list_container.innerHTML = ''; // Clear existing list

          if (filteredTasks.length === 0) {
              this.renderEmptyState();
              return;
          }

          filteredTasks.forEach(task => {
              const taskElement = this.createTaskElement(task);
              this.task_list_container.appendChild(taskElement);
          });
      }

      createTaskElement(task) {
          const li = document.createElement('li');
          li.className = `task-item ${task.completed ? 'completed' : ''}`;
          li.dataset.taskId = task.id;

          const checkbox = document.createElement('input');
          checkbox.type = 'checkbox';
          checkbox.checked = task.completed;
          checkbox.addEventListener('change', () => this.toggleTaskComplete(task.id));

          const title = document.createElement('span');
          title.className = 'task-title';
          title.textContent = task.title;

          const deleteButton = document.createElement('button');
          deleteButton.className = 'delete-btn';
          deleteButton.textContent = 'Delete';
          deleteButton.addEventListener('click', () => this.deleteTask(task.id));

          li.appendChild(checkbox);
          li.appendChild(title);
          li.appendChild(deleteButton);

          return li;
      }

      renderEmptyState() {
          this.task_list_container.innerHTML = '<p class="empty-state">No tasks yet. Add one above!</p>';
      }


      // Event Handling
      bindEventListeners() {
          if (this.quick_add_form) {
              this.quick_add_form.addEventListener('submit', (e) => {
                  e.preventDefault();
                  const title = this.quick_add_input.value;
                  this.createTask(title);
                  this.quick_add_input.value = '';
              });
          }

          if (this.search_input) {
              this.search_input.addEventListener('input', (e) => {
                  this.setSearchQuery(e.target.value);
              });
          }

          if (this.filter_tabs) {
              this.filter_tabs.addEventListener('click', (e) => {
                  if (e.target.dataset.filter) {
                      this.setFilter(e.target.dataset.filter);
                  }
              });
          }
      }

      // Actions
      createTask(title) {
          if (!title || title.trim() === '') return;

          const now = new Date().toISOString();
          const newTask = {
              id: self.crypto.randomUUID(),
              title: title.trim(),
              notes: '',
              completed: false,
              createdAt: now,
              updatedAt: now,
              completedAt: null
          };
          this.state.tasks.unshift(newTask);
          this.saveState();
          this.render();
      }

      toggleTaskComplete(taskId) {
          const task = this.state.tasks.find(t => t.id === taskId);
          if (task) {
              task.completed = !task.completed;
              task.updatedAt = new Date().toISOString();
              task.completedAt = task.completed ? task.updatedAt : null;
              this.saveState();
              this.render();
          }
      }

      deleteTask(taskId) {
          this.state.tasks = this.state.tasks.filter(t => t.id !== taskId);
          this.saveState();
          this.render();
      }

      updateTask(updatedTask) {
          const index = this.state.tasks.findIndex(t => t.id === updatedTask.id);
          if (index !== -1) {
              this.state.tasks[index] = { ...this.state.tasks[index], ...updatedTask, updatedAt: new Date().toISOString() };
              this.saveState();
              this.render();
          }
      }

      setFilter(filter) {
          this.state.ui.activeFilter = filter;
          this.render();
      }

      setSearchQuery(query) {
          this.state.ui.searchQuery = query;
          this.render();
      }

      // Computed Properties
      getFilteredTasks() {
          const { tasks, ui } = this.state;

          let filtered = tasks;

          // Apply filter
          if (ui.activeFilter === 'active') {
              filtered = tasks.filter(task => !task.completed);
          } else if (ui.activeFilter === 'completed') {
              filtered = tasks.filter(task => task.completed);
          }

          // Apply search
          if (ui.searchQuery) {
              const searchTerm = ui.searchQuery.toLowerCase();
              filtered = filtered.filter(task =>
                  task.title.toLowerCase().includes(searchTerm) ||
                  (task.notes && task.notes.toLowerCase().includes(searchTerm))
              );
          }

          // Apply sorting
          return filtered.sort((a, b) => new Date(b.updatedAt) - new Date(a.updatedAt));
      }
